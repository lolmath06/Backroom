package com.dolmen.backroom.entity.custom;

import com.dolmen.backroom.entity.ModEntities;
import net.minecraft.core.BlockPos;
import net.minecraft.core.Direction;
import net.minecraft.server.level.ServerLevel;
import net.minecraft.world.damagesource.DamageSource;
import net.minecraft.world.entity.*;
import net.minecraft.world.entity.ai.attributes.AttributeSupplier;
import net.minecraft.world.entity.ai.attributes.Attributes;
import net.minecraft.world.entity.ai.control.FlyingMoveControl;
import net.minecraft.world.entity.ai.goal.*;
import net.minecraft.world.entity.ai.goal.target.HurtByTargetGoal;
import net.minecraft.world.entity.ai.navigation.FlyingPathNavigation;
import net.minecraft.world.entity.ai.navigation.PathNavigation;
import net.minecraft.world.entity.monster.Monster;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.level.Level;
import net.minecraft.world.level.block.state.BlockState;
import net.minecraft.nbt.CompoundTag;
import net.minecraft.network.syncher.EntityDataAccessor;
import net.minecraft.network.syncher.EntityDataSerializers;
import net.minecraft.network.syncher.SynchedEntityData;
import org.apache.logging.log4j.LogManager;
import org.apache.logging.log4j.Logger;
import org.jetbrains.annotations.Nullable;

public class DeathMothsEntity extends Monster {
    private static final Logger LOGGER = LogManager.getLogger();

    public final AnimationState idleGroundAnimationState = new AnimationState();
    public final AnimationState idleWallAnimationState = new AnimationState();
    public final AnimationState flyAnimationState = new AnimationState();
    public final AnimationState attackAnimationState = new AnimationState();

    public enum DeadMothAnimation {
        IDLE_GROUND, IDLE_WALL, FLY, ATTACK
    }

    private DeadMothAnimation currentAnimation = DeadMothAnimation.IDLE_GROUND;
    private int animationStartTick = 0;

    private static final EntityDataAccessor<Boolean> DATA_STICKING_TO_WALL =
            SynchedEntityData.defineId(DeathMothsEntity.class, EntityDataSerializers.BOOLEAN);

    private boolean stickingToWall = false;

    private int attackTicksRemaining = 0;
    private static final byte ATTACK_EVENT = 69;

    private int aggressiveTicks = 0;

    public DeathMothsEntity(EntityType<? extends Monster> type, Level level) {
        super(type, level);
        this.moveControl = new FlyingMoveControl(this, 10, true);
        this.setNoGravity(true);
    }

    @Override
    protected void defineSynchedData(SynchedEntityData.Builder builder) {
        super.defineSynchedData(builder);
        builder.define(DATA_STICKING_TO_WALL, false);
    }

    @Override
    protected PathNavigation createNavigation(Level level) {
        return new FlyingPathNavigation(this, level);
    }

    @Override
    public void aiStep() {
        super.aiStep();
        this.setNoGravity(true);

        if (aggressiveTicks > 0) {
            aggressiveTicks--;
            if (aggressiveTicks == 0) {
                LOGGER.info("[DENMOS] L'agressivité s'est calmée.");
            }
        }
    }

    public static AttributeSupplier.Builder createAttributes() {
        return Monster.createMonsterAttributes()
                .add(Attributes.MAX_HEALTH, 20.0D)
                .add(Attributes.MOVEMENT_SPEED, 0.30D)
                .add(Attributes.FOLLOW_RANGE, 24.0D)
                .add(Attributes.ATTACK_DAMAGE, 4.0D)
                .add(Attributes.FLYING_SPEED, 0.35D);
    }

    @Override
    protected void registerGoals() {
        this.goalSelector.addGoal(0, new StickToWallGoal(this));
        this.goalSelector.addGoal(1, new MeleeAttackGoal(this, 1.2, true));
        this.goalSelector.addGoal(2, new FlyAroundGoal(this));
        this.goalSelector.addGoal(3, new ConditionalLookAtPlayerGoal(this, Player.class, 8.0F));
        this.goalSelector.addGoal(4, new ConditionalRandomLookAroundGoal(this));
    }

    // Custom goal qui ne regarde le joueur que si pas collé au mur
    static class ConditionalLookAtPlayerGoal extends LookAtPlayerGoal {
        private final DeathMothsEntity mob;
        public ConditionalLookAtPlayerGoal(DeathMothsEntity mob, Class<? extends LivingEntity> lookAtType, float lookDistance) {
            super(mob, lookAtType, lookDistance);
            this.mob = mob;
        }
        @Override
        public boolean canUse() {
            return !mob.isStickingToWall() && super.canUse();
        }
    }
    // Custom goal qui ne regarde aléatoirement que si pas collé au mur
    static class ConditionalRandomLookAroundGoal extends RandomLookAroundGoal {
        private final DeathMothsEntity mob;
        public ConditionalRandomLookAroundGoal(DeathMothsEntity mob) {
            super(mob);
            this.mob = mob;
        }
        @Override
        public boolean canUse() {
            return !mob.isStickingToWall() && super.canUse();
        }
    }

    public DeadMothAnimation getCurrentAnimation() {
        return currentAnimation;
    }

    public void setCurrentAnimation(DeadMothAnimation animation) {
        if (currentAnimation != animation) {
            LOGGER.info("[DENMOS] Animation changée : {} --> {}", currentAnimation, animation);
            currentAnimation = animation;
            animationStartTick = this.tickCount;
        }
    }

    private void updateAnimationStates() {
        int elapsed = this.tickCount - animationStartTick;
        switch (currentAnimation) {
            case IDLE_GROUND -> {
                if (!idleGroundAnimationState.isStarted()) idleGroundAnimationState.start(animationStartTick);
                idleWallAnimationState.stop();
                flyAnimationState.stop();
                attackAnimationState.stop();
            }
            case IDLE_WALL -> {
                idleGroundAnimationState.stop();
                if (!idleWallAnimationState.isStarted()) idleWallAnimationState.start(animationStartTick);
                flyAnimationState.stop();
                attackAnimationState.stop();
            }
            case FLY -> {
                idleGroundAnimationState.stop();
                idleWallAnimationState.stop();
                if (!flyAnimationState.isStarted()) flyAnimationState.start(animationStartTick);
                attackAnimationState.stop();
            }
            case ATTACK -> {
                idleGroundAnimationState.stop();
                idleWallAnimationState.stop();
                flyAnimationState.stop();
                if (!attackAnimationState.isStarted()) attackAnimationState.start(animationStartTick);
            }
        }
    }

    public boolean isFlying() {
        if (isStickingToWall()) return false;
        return !this.onGround();
    }

    public boolean isStickingToWall() {
        return this.entityData.get(DATA_STICKING_TO_WALL);
    }

    public void setStickingToWall(boolean stick) {
        LOGGER.info("[DENMOS] setStickingToWall({}) à {}", stick, this.blockPosition());
        if (!this.level().isClientSide) {
            this.stickingToWall = stick;
            this.entityData.set(DATA_STICKING_TO_WALL, stick);
        }
    }

    public boolean isAggressive() {
        return aggressiveTicks > 0;
    }
    public void makeAggressive(int duration) {
        aggressiveTicks = duration;
        LOGGER.info("[DENMOS] MOB AGRESSE pendant {} ticks", duration);
        if (isStickingToWall()) {
            setStickingToWall(false);
            LOGGER.info("[DENMOS] Décollage du mur suite à agression !");
        }
    }

    @Override
    public boolean hurt(DamageSource source, float amount) {
        boolean result = super.hurt(source, amount);
        if (!this.level().isClientSide) {
            makeAggressive(100);
        }
        return result;
    }

    @Nullable
    public BlockPos findWallToStick() {
        BlockPos pos = this.blockPosition();
        Level level = this.level();
        LOGGER.info("[DENMOS][DEBUG] findWallToStick() called from {}", pos);

        for (Direction dir : Direction.Plane.HORIZONTAL) {
            for (int dist = 1; dist <= 3; dist++) {
                for (int yOffset = -2; yOffset <= 2; yOffset++) {
                    BlockPos wallBase = pos.relative(dir, dist).offset(0, yOffset, 0);
                    boolean is4High = true;

                    for (int y = 0; y < 4; y++) {
                        BlockPos check = wallBase.above(y);
                        BlockState state = level.getBlockState(check);
                        LOGGER.info("[DENMOS][DEBUG] Test bloc pour mur: {} (Yoffset={} y={}), état: {} (isAir={})", check, yOffset, y, state.getBlock(), state.isAir());
                        if (state.isAir()) {
                            is4High = false;
                            break;
                        }
                    }
                    LOGGER.info("[DENMOS][DEBUG] Test de colonne dir={} dist={} Yoffset={} : wallBase={} => is4High={}", dir, dist, yOffset, wallBase, is4High);

                    if (is4High) {
                        BlockPos stickTo = wallBase.above(1);
                        BlockPos inFront = stickTo.relative(dir.getOpposite());
                        BlockPos inFront2 = stickTo.relative(dir.getOpposite(), 2);
                        boolean frontAir = level.getBlockState(inFront).isAir();
                        boolean front2Air = level.getBlockState(inFront2).isAir();
                        LOGGER.info("[DENMOS][DEBUG] inFront={} isAir={} | inFront2={} isAir={}", inFront, frontAir, inFront2, front2Air);

                        if (frontAir && front2Air) {
                            LOGGER.info("[DENMOS] Mur de 4 trouvé à {} dir={} | Base: {} | baseY={} | inFront: {} et {}", pos, dir, wallBase, wallBase.getY(), inFront, inFront2);
                            return stickTo;
                        } else {
                            LOGGER.info("[DENMOS][DEBUG] Mur détecté mais pas assez d'air devant : {} ou {}", inFront, inFront2);
                        }
                    }
                }
            }
        }
        LOGGER.info("[DENMOS] Aucun mur de 4 à proximité depuis {}", pos);
        return null;
    }

    static class StickToWallGoal extends Goal {
        private final DeathMothsEntity mob;
        private BlockPos wallPos;

        public StickToWallGoal(DeathMothsEntity mob) { this.mob = mob; }

        @Override
        public boolean canUse() {
            if (mob.isStickingToWall()) return false;
            if (mob.isAggressive()) return false;
            wallPos = mob.findWallToStick();
            boolean result = wallPos != null;
            mob.LOGGER.info("[DENMOS] StickToWallGoal.canUse = {} (wallPos = {})", result, wallPos);
            return result;
        }

        @Override
        public void start() {
            mob.setStickingToWall(true);
            mob.getNavigation().moveTo(wallPos.getX() + 0.5, wallPos.getY() + 0.1, wallPos.getZ() + 0.5, 1.2);
            mob.setCurrentAnimation(DeadMothAnimation.IDLE_WALL);
            mob.LOGGER.info("[DENMOS] START COLLAGE MUR à {}", wallPos);
        }

        @Override
        public boolean canContinueToUse() {
            boolean stillStuck = mob.isStickingToWall() && !mob.isAggressive();
            mob.LOGGER.info("[DENMOS] StickToWallGoal.canContinueToUse = {} | stickingToWall={} | aggressive={}", stillStuck, mob.isStickingToWall(), mob.isAggressive());
            return stillStuck;
        }

        @Override
        public void tick() {
            double dist = mob.position().distanceTo(net.minecraft.world.phys.Vec3.atCenterOf(wallPos));
            if (dist > 0.2) {
                mob.getNavigation().moveTo(wallPos.getX() + 0.5, wallPos.getY() + 0.1, wallPos.getZ() + 0.5, 1.2);
                mob.LOGGER.info("[DENMOS] Correction position collage mur : {} -> {}", mob.position(), wallPos);
            } else {
                mob.setDeltaMovement(0, 0, 0);
                mob.getNavigation().stop();
                mob.setPos(wallPos.getX() + 0.5, wallPos.getY() + 0.1, wallPos.getZ() + 0.5);
            }
            mob.setCurrentAnimation(DeadMothAnimation.IDLE_WALL);
        }

        @Override
        public void stop() {
            mob.setStickingToWall(false);
            mob.LOGGER.info("[DENMOS] STOP COLLAGE MUR (à {}), on repasse à l'état normal", wallPos);
        }
    }

    static class FlyAroundGoal extends Goal {
        private final DeathMothsEntity mob;
        public FlyAroundGoal(DeathMothsEntity mob) { this.mob = mob; }
        @Override
        public boolean canUse() { return !mob.isStickingToWall(); }
        @Override
        public void start() { pickNewDestination(); }
        @Override
        public void tick() { if (mob.getNavigation().isDone()) { pickNewDestination(); } }
        private void pickNewDestination() {
            double x = mob.getX() + (mob.getRandom().nextDouble() - 0.5) * 8.0;
            double y = Math.min(mob.getY() + (mob.getRandom().nextDouble() - 0.5) * 4.0 + 2.0,
                    mob.level().getMaxBuildHeight() - 2);
            double z = mob.getZ() + (mob.getRandom().nextDouble() - 0.5) * 8.0;
            BlockPos pos = new BlockPos((int) x, (int) (y + 1), (int) z);
            if (!mob.level().getBlockState(pos).isAir()) y -= 1.0;
            mob.getNavigation().moveTo(x, y, z, 1.0);
            mob.LOGGER.info("[DENMOS] FlyAroundGoal: nouvelle destination {},{},{}", x, y, z);
        }
    }

    @Override
    public boolean doHurtTarget(Entity target) {
        if (!this.level().isClientSide) {
            this.level().broadcastEntityEvent(this, ATTACK_EVENT);
            LOGGER.info("[DENMOS] Attaque cible: {} à {}", target, this.blockPosition());
        }
        return super.doHurtTarget(target);
    }

    @Override
    public void handleEntityEvent(byte id) {
        if (id == ATTACK_EVENT) {
            this.attackTicksRemaining = 10;
            LOGGER.info("[DENMOS] handleEntityEvent: attaque déclenchée !");
        } else {
            super.handleEntityEvent(id);
        }
    }

    @Override
    public boolean causeFallDamage(float fallDistance, float damageMultiplier, DamageSource source) {
        return false;
    }

    @Override
    public void tick() {
        super.tick();

        LOGGER.info("[DENMOS] TICK - Animation: {} | stickingToWall: {} | aggressiveTicks: {} | pos: {} | attackTicks: {}", currentAnimation, isStickingToWall(), aggressiveTicks, this.blockPosition(), attackTicksRemaining);

        if (this.isStickingToWall()) {
            // Fige complètement la rotation (statue sur le mur)
            this.setYRot(this.getYRot());
            this.setXRot(0.0f);
            this.yBodyRot = this.getYRot();
            this.yHeadRot = this.getYRot();
            this.setDeltaMovement(0, 0, 0);
        }

        if (this.level().isClientSide) {
            updateAnimationStates();
        }

        if (attackTicksRemaining > 0) {
            setCurrentAnimation(DeadMothAnimation.ATTACK);
            attackTicksRemaining--;
        } else if (isFlying()) {
            setCurrentAnimation(DeadMothAnimation.FLY);
        } else if (isStickingToWall()) {
            setCurrentAnimation(DeadMothAnimation.IDLE_WALL);
        } else if (this.onGround()) {
            setCurrentAnimation(DeadMothAnimation.IDLE_GROUND);
        }
    }

    @Override
    public void addAdditionalSaveData(CompoundTag tag) {
        super.addAdditionalSaveData(tag);
        tag.putBoolean("StickingToWall", this.isStickingToWall());
    }

    @Override
    public void readAdditionalSaveData(CompoundTag tag) {
        super.readAdditionalSaveData(tag);
        boolean stick = tag.getBoolean("StickingToWall");
        if (!this.level().isClientSide) {
            this.stickingToWall = stick;
            this.entityData.set(DATA_STICKING_TO_WALL, stick);
        }
    }

}
